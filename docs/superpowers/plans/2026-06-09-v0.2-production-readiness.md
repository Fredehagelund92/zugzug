## Task A5: Persist `scan_run` rows (#7 part 2)

**Files:**

### Step 1: Wrap job execution

In `scheduler.ts`, the tick logic now:

```ts
for (const job of jobs) {
  const runId = `run_${crypto.randomUUID().replace(/-/g, "")}`;
  const startedAt = new Date();
  await pgRun(`INSERT INTO ${pg("scan_run")} (id, source_id, started_at, status) VALUES ($1, $2, $3, 'running')`, [runId, job.name, startedAt]);
  try {
    const result = await job.run(ctx);
    await pgRun(`UPDATE ${pg("scan_run")} SET ended_at = $1, status = 'ok', rows_scanned = $2, duration_ms = $3 WHERE id = $4`, [new Date(), result.rowsScanned ?? null, Date.now() - startedAt.getTime(), runId]);
  } catch (err) {
    await pgRun(`UPDATE ${pg("scan_run")} SET ended_at = $1, status = 'error', error_message = $2, duration_ms = $3 WHERE id = $4`, [new Date(), err.message, Date.now() - startedAt.getTime(), runId]);
    // Don't re-throw — other jobs in the tick should still run.
  }
}
```

Note: `source_id` here is the JOB NAME for scheduler-level runs. For source-scoped scan jobs (where multiple sources scan per tick), use the actual source ID. Adjust the schema if needed.

### Step 1: Define the helper

In `auth.ts`:

```ts
export type Role = "admin" | "editor" | "viewer";
export type Operation =
  | "curate"       // create/update drafts
  | "commit"       // commit drafts to canonical
  | "manage_team"  // change roles, manage allowlist
  | "manage_adapter"; // configure warehouse credentials

export function canMutate(role: Role, op: Operation): boolean {
  const matrix: Record<Role, Operation[]> = {
    admin: ["curate", "commit", "manage_team", "manage_adapter"],
    editor: ["curate", "commit"],
    viewer: [],
  };
  return matrix[role].includes(op);
}
```

```ts
const baseShared = z.object({
  account: z.string(),
  warehouse: z.string(),
  database: z.string(),
  schema: z.string(),
  role: z.string().optional(),
});

const keyPair = baseShared.extend({
  authMode: z.literal("key-pair"),
  username: z.string(),
  privateKeyPem: z.string(),  // existing
  privateKeyPassphrase: z.string().optional(),
});

const password = baseShared.extend({
  authMode: z.literal("password"),
  username: z.string(),
  password: z.string(),
});

const externalBrowser = baseShared.extend({
  authMode: z.literal("external-browser"),
  username: z.string(),
});

export const SnowflakeCredentials = z.discriminatedUnion("authMode", [keyPair, password, externalBrowser]);
export type SnowflakeCredentials = z.infer<typeof SnowflakeCredentials>;
```

```ts
function buildConnectionConfig(creds: SnowflakeCredentials): ConnectionOptions {
  const base = {
    account: creds.account,
    warehouse: creds.warehouse,
    database: creds.database,
    schema: creds.schema,
    role: creds.role,
    username: creds.username,
  };
  switch (creds.authMode) {

}
```
